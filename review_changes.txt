MODIFICACIÓN: Estandarización de autenticación mediante la cookie 'pb_auth' y manejo robusto de tokens Bearer.

```typescript
export async function initPocketBase(req: NextRequest): Promise<PocketBase> {
  const client = getPocketBase();
  client.authStore.clear();

  const authValue = req.cookies.get('pb_auth')?.value;
  const authHeader = req.headers.get('Authorization');

  if (authHeader?.startsWith('Bearer ')) {
    const token = authHeader.substring(7);
    client.authStore.save(token, null);
  } else if (authValue) {
    try {
      if (authValue.includes('%7B')) {
        client.authStore.loadFromCookie('pb_auth=' + authValue);
      } else {
        client.authStore.save(authValue, null);
      }
    } catch (_) {
      client.authStore.save(authValue, null);
    }
  }
  // ... (refresh logic)
  return client;
}
```

    const adminPb = new PocketBase(POCKETBASE_URL);
    await adminPb.admins.authWithPassword(email, password);
